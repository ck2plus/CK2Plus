trade_post = {
	trigger = {

		# Game Rule Limits
		# Coast: Only MRs can build on coastal provinces, non-MRs can only build on inland trade route provinces
		trigger_if = {
			limit = {
				has_game_rule = {
					name = tp_construction
					value = mr_coast
				}

				FROM = { is_patrician = yes }
			}

			trigger_if = {
				limit = { FROM = { is_patrician = yes } }
				is_coastal = yes
			}
			trigger_else = {
				is_coastal = no
			}
		}
		# Only Merchants: Only MRs can build Trade Posts
		trigger_else_if = {
			limit = {
				has_game_rule = { 	# Only MRs are allowed to build any TPs
					name = tp_construction
					value = only_mr
				}
			}

			FROM = { is_patrician = yes }
		}


		# Location Limits
		# The distance/cost calculation is broken for the Caspian being far too cheap, so we'll hack in a lock
		trigger_if = { # Only those in the Caspian can build in the Caspian
			limit = {
				is_coastal = yes
				any_neighbor_province = { region = caspian_sea_zone }
			}

			custom_tooltip = {
				text = TOOLTIP_NO_CASPIAN_TPS # Tooltip may need refined
				FROM = {
					top_liege = {
						any_realm_province = {
							any_neighbor_province = { region = caspian_sea_zone  }
						}
					}
				}
			}
		}

		is_within_diplo_range = FROM # No you can't build a TP half way across the world just because you want to.
	}
}
