religion_has_haruspicy_trigger = {
	trigger_if = {
		limit = { has_dlc = "Holy Fury" }

		OR = {
			has_religion_feature = religion_feature_hellenic
			has_religion_feature = religion_haruspicy

			# CK2Plus
			has_religion_feature = religion_feature_celtic

			trigger_if = {
				limit = { religion = hellenic_pagan }

				NOR = {
					has_alternate_start_parameter = {
						key = religion_features
						value = random
					}

					has_alternate_start_parameter = {
						key = religion_names
						value = random
					}
				}
			}
		}
	}
	trigger_else = {
		religion_openly_hellenic_or_reformed_trigger = yes
	}
}

religion_is_adventuring_and_seafaring_trigger = {
	trigger_if = {
		limit = { has_dlc = "Holy Fury" }

		OR = {
			has_religion_feature = religion_feature_norse

			AND = {
				has_religion_feature = religion_adventuring

				OR = {
					has_religion_feature = religion_seafaring
					has_religion_feature = religion_feature_aztec
				}
			}

			trigger_if = {
				limit = { religion = norse_pagan }

				NOR = {
					has_global_flag = random_religion_names
					has_global_flag = random_religion_features
				}
			}
		}
	}
	trigger_else = {
		religion_openly_norse_or_reformed_trigger = yes
	}
}
