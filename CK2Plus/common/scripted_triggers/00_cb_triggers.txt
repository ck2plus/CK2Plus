suitable_holdings_trigger = {
	trigger_if = {
		limit = {
			ROOT = { is_nomadic = no }
			FROM = { is_nomadic = yes }
		}

		any_direct_de_jure_vassal_title = {
			any_direct_de_jure_vassal_title = {
				OR = {
					holding_type = castle
					holding_type = city

					trigger_if = {
						limit = { holding_type = tribal }
						ROOT = { is_tribal = yes }
					}
				}
			}
		}
	}
}

temujin_avoid_china_trigger = {
	trigger_if = {
		limit = {
			ROOT = {
				has_character_flag = is_temujin
				has_landed_title = e_mongol_empire
				is_landed = yes
			}
		}

		FROM = {
			NOT = { has_landed_title = e_china_west_governor }
		}
	}
}
